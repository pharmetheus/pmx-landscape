The FDA reviewer’s independent analysis was consistent with the Applicant’s findings
demonstrating a decrease in camizestrant AUC and Cmax GMR to 30% and 40%,
respectively, when administered concomitantly with carbamazepine.

FDA does not agree that moderate CYP3A inducers may be administered concomitantly
with camizestrant with caution. FDA recommends avoiding concomitant use of
camizestrant and moderate inducers of CYP3A; however, PBPK simulations support
increasing the dosage of camizestrant to 150 mg QD when given in combination with
palbociclib or abemaciclib if concomitant use with a moderate CYP3A inducer cannot be
avoided. Based on the PBPK modeling, the simulated exposures of camizestrant in
patients with cancer when camizestrant 150 mg QD is co-administered with a moderate
CYP3A inducer (phenobarbital or efavirenz) are similar to the observed geometric mean
Cmaxss of 74 ng/mL and AUC, of 1,116 ng/mL-h of camizestrant on Day 15 of Cycle 1 in
patients following administration of camizestrant 75 mg QD in SERENA-1. Therefore,
the dose of camizestrant may be increased to 150 mg QD when given in combination
with palbociclib or abemaciclib if concomitant use with a moderate CYP3A inducer
cannot be avoided. FDA does not recommend an increase in camizestrant dosage for
patients receiving camizestrant in combination with ribociclib concomitantly with a
moderate CYP3A inducer, as the exposure of camizestrant 75 mg QD when combined
with ribociclib is already increased 2-fold. See Section 19.4.3.4.2 for detailed PBPK
assessment.
